B. Rhythmic String

称长度为 m 的二进制串是节奏的，当且仅当它有周期 p 且前 p 个字符中恰有一个 1。

现在给定数组 a[1…n] 和一个字符串 s，初始时由 1018 个零组成。对每个元素 ai，字符串中所有索引能被 ai 整除的字符都被替换为 1。现在请问至少从 a 中移除多少元素能够让 s 变为节奏的串。

考虑最终保留下来数中最小值为 d。则：

sd=s2d=⋯=1

前 p 个字符中恰有一个 1，则必然 d≤p<2d；又由于每个周期只有一个 1，则显然 d 和 2d 应当属于相同的模 p 的余数类，进而得到：

p=d

进而字符串中所有为 1 的位置都是 d,2d,3d,… 而保留下来的所有 aj 的位置及其倍数都得是 1，显然需要 d|aj。定义：

f(d)=#{i:d|ai}

答案就是 n−maxd∈af(d)。本题瓶颈主要在于证明 d=p，剩下的操作无非是运用筛法的一些思想来求 maxd∈af(d)。

#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int M=31623;
vector<int> pri;
void init(){
    vector<int> vis(M+1);
    for(int i=2;i<=M;i++){
        if(vis[i])continue;
        pri.push_back(i);
        for(int j=i*i;j<=M;j+=i)vis[j]=1;
    }
}
vector<int> getdiv(int x){
    vector<int> d={1};
    for(int p:pri){
        if(1ll*p*p>x)break;
        if(x%p)continue;
        int sz=d.size(),t=1;
        while(x%p==0){
            x/=p;
            t*=p;
            for(int i=0;i<sz;i++)
                d.push_back(d[i]*t);
        }
    }
    if(x>1){
        int sz=d.size();
        for(int i=0;i<sz;i++)
            d.push_back(d[i]*x);
    }
    return d;
}
void solve(){
    int n;
    cin>>n;
    vector<int> a(n+1);
    unordered_map<int,int> cnt,f;
    for(int i=1;i<=n;i++){
        cin>>a[i];
        cnt[a[i]]++;
    }
    sort(a.begin()+1,a.end());
    a.erase(unique(a.begin()+1,a.end()),a.end());
    int ans=0;
    for(int i=1;i<(int)a.size();i++){
        vector<int> d=getdiv(a[i]);
        for(int j=0;j<(int)d.size();j++){
            int x=d[j];
            if(cnt.count(x))
                f[x]+=cnt[a[i]];
        }
    }
    for(int i=1;i<(int)a.size();i++)
        ans=max(ans,f[a[i]]);
    cout<<n-ans<<"\n";
}
int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    init();
    int T;
    cin>>T;
    while(T--)solve();
    return 0;
}

C. Big XOR

给出正整数 n，其生成一个数组 a=[1,2,…,n]。

记 k 为 a 所有元素按位或的结果，现在你可以选择 a 中某些位置将其对应的数值翻倍（每个位置只能操作一次），设 m 为操作完毕后的 a 所有元素的按位异或的结果。

请你给出一种选择位置的方案，使得 m≥2k。

或操作把能变成 1 的位置都变成 1。考虑 n 是 t 位二进制数，也即

2t−1≤n<2t

显然有 k=2t−1，也即 k 的二进制表达应该是 (11…1)2

于是 2k=(11…10)2，为 t+1 位二进制数。符合条件的 m ，必须满足第 1 位到第 t 位全部为 1。我们期望操作最好是单独影响一个或两个位，所以考虑对某个 2i 翻倍，则 XOR 变化如：

m′=m⊕2i⊕2i+1

其实质就是翻转 m 的第 i 位和第 i+1 位，根据这个可以从高位向低位贪心：如果第 t 位是 0，就翻倍 2t−1；进而继续检查下去，直到第一位。

#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
void solve(){
    int n;
    cin>>n;
    vector<int> a(n+1,0);
    ll m=0,p=1;
    for(int i=1;i<=n;i++)m^=i;
    while(p<=n)p<<=1;
    for(p>>=1;p;p>>=1)
        if(!(m&(p<<1)))a[p]=1,m^=p^(p<<1);
    for(int i=1;i<=n;i++)cout<<a[i]<<" ";
    cout<<"\n";
}
int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin>>t;
    while(t--)solve();
    return 0;
}

D. Black, Gray and White

给出一个编号从 1 到 n 的树，初始时所有顶点都是白色。

选择一个整数 k(1≤k≤2n)，对从 1 到 n 的每个 i，若树中存在编号 k−i 的顶点且 k−i≠i，则：

  • 顶点 i,k−i 变化为黑色。

  • 顶点 i→k−i 的简单路径上所有白色节点被变化为灰色。

试问如何选择 k 以最大化 pair<black,gray>？

先确保黑色点最多，在该前提下使灰色点最多。由于黑色点的个数必然是偶数，对 n 按照奇偶性分类讨论：

如果 n 是偶数，令 k=n+1，从而原树形成了：

(1,n) (2,n−1)…(n/2,n/2+1)

的两两配对，全部变黑。

对于 n 为奇数，最多有 n−1 个黑色点。只有 k=n,n+1,n+2 能够使黑色点最多，分别对应失配节点编号为 n,(n+1)/2,1。我们希望让失配节点变灰。

一个失配点 r 变灰，等价于是否存在配对 (i,j) 使 r∈path(i,j)。若以 r 为根，则 r 有若干子树。通过 dfs 可以知道某个节点 u 对应的子树根节点 bel[u]，而：

r∈path(i,j)⇔bel[i]≠bel[j]

只需检验固定 k 下的所有配对 (i,k−i) 是否满足 bel[i]≠bel[k−i] 即可。

        int u,v;
        cin>>u>>v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    if(!(n&1)){
        cout<<n+1<<"\n";
        return;
    }
    if(n==1){
        cout<<1<<"\n";
        return;
    }
    auto check=[&](int k,int rt){
        vector<int> fa(n+1),bel(n+1);
        vector<int> q={rt};
        for(int i=0;i<(int)q.size();i++){
            int u=q[i];
            for(int v:g[u]){
                if(v==fa[u])continue;
                fa[v]=u;
                bel[v]=(u==rt ? v:bel[u]);
                q.push_back(v);
            }
        }
        for(int i=1;i<=n;i++){
            int j=k-i;
            if(j<1 or j>n or i>=j)continue;

        }
        return 0;
    };
    if(check(n,n))
        cout<<n;
    else if(check(n+1,(n+1)/2))

E. Partition of a Binary String

给定一个长度为 n 的二进制串 s，下标从 1 开始到 n。

将集合 {1,2,…,n} 划分成若干个子集。同子集的任意一对不同整数 x,y，需要满足下述两条件之一：

  • sx=sy

  • |x−y|=1

对一个子集 M，令 f(M) 表示 x∈M 中不同的 sx 数量，定义代价：

cost(M)=|M|⋅f(M)

划分的代价是所有子集代价的线性和。

问对 s 划分的最大代价以及等同该代价的划分数量。由于结果可能很大，对 998244353 取模。

这个题看着难，但“任意”这个条件特别强，从这里入手分析 M 的简单结构。

第一类合法子集 M1 包含的所有下标对应的字符都相同，则 cost(M1)=|M1|

第二类合法子集 M2 同时包含 0/1 ，则任意不同字符位置必须相邻，只可能是下标连续段且对应 01/10 或 010/101 的。 cost(M2)=2|M2|

设所有的第二类合法子集总共包含了 t 个位置，总代价就是

2t+(n−t)=n+t

故最大化总代价就是最大化 t，也即用互不相交的 01/10/010/101 尽可能多的覆盖字符串位置。将字符串 s 拆分成若干极大交替段，例如

s=01010|0|101|1

对每个段如果 len≥2，显然任何这样的 len 都可由 2 与 3 的线性组合表达，必然能合法覆盖；对于 len=1 的无法被覆盖，设这样的点段有 c 个，总代价即为

2n−c

最优化问题解决，下面考虑方案数。这是经典的计数 dp。令 f[i] 表示长度为 i 的交替段全部划分为第二类合法集合的方案数，显然

f[i]=f[i−2]+f[i−3],∀i≥3

边界为 f[0]=1,f[1]=0,f[2]=1。如此解决了第二类的划分。

而长度为 1 的独立段由 k0 个 0 和 k1 个 1 组成，它们可以任意分组到第一类对应集合里去。可以套用集合划分计数模型，也即 bell 数。

最终方案数：

Bk0Bk1∏1≤i<q,leni≥2f(leni)(mod998244353)

#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod=998244353,N=5060;
int f[N],bell[N];
void prework(){
    f[0]=1;f[1]=0;f[2]=1;
    for(int i=3;i<N;i++) f[i]=(f[i-2]+f[i-3])%mod;
    vector<int> dp(N);
    dp[0]=bell[0]=1;
    for(int i=1;i<N;i++){
        for(int j=i;j>=1;j--){
            dp[j]=(dp[j-1]+1ll*j*dp[j])%mod;
            bell[i]=(bell[i]+dp[j])%mod;
        }
        dp[0]=0;
    }
}
void solve(){
    int n;string s;
    cin>>n>>s;
    int k[2]={0,0},ans=1;
    for(int i=0;i<n;){
        int j=i+1;
        while(j<n && s[j]!=s[j-1])j++;
        int len=j-i;
        if(len==1)k[s[i]-'0']++;
        else ans=1ll*ans*f[len]%mod;
        i=j;
    }
    ans=1ll*ans*bell[k[0]]%mod;
    ans=1ll*ans*bell[k[1]]%mod;
    cout<<2*n-k[0]-k[1]<<" "<<ans<<"\n";
}


int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    prework();
    int t;
    cin>>t;
    while(t--)solve();
    return 0;
}
